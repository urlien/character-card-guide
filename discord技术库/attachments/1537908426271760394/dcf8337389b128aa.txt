<script>
  // -------- 工具函数 --------
  function escapeHtml(text) {
    const div = document.createElement('div');
    div.textContent = text;
    return div.innerHTML;
  }

  // -------- 数据 --------
  let roleTagList = JSON.parse(localStorage.getItem('roleTags')) || ['主角', '配角', '反派'];
  let selectedRoles = JSON.parse(localStorage.getItem('selectedRoles')) || [];
  let plotTagList = JSON.parse(localStorage.getItem('plotTags')) || ['战斗', '对话', '伏笔'];

  const S = (k, v) => localStorage.setItem(k, JSON.stringify(v));

  // -------- 通用：从文本域解析多行标签 --------
  function parseTagsFromTextarea(textarea) {
    return textarea.value.split('\n').map(s => s.trim()).filter(s => s !== '');
  }

  // -------- 渲染 --------
  function render() {
    // 渲染角色标签
    const rt = document.getElementById('roleTags');
    rt.innerHTML = roleTagList.map((tag, i) => {
      const sel = selectedRoles.includes(tag) ? 'selected' : '';
      const safeTag = escapeHtml(tag);
      return `<span class="xx-tag ${sel}" data-type="role" data-index="${i}" data-tag="${tag}">
        <span class="xx-tag-text">${safeTag}</span>
        <span class="xx-tag-edit" data-action="edit">✎</span>
        <span class="xx-tag-del" data-action="delete">×</span>
      </span>`;
    }).join('');

    // 渲染已选角色列表
    const sl = document.getElementById('selectedRoles');
    if (selectedRoles.length === 0) {
      sl.innerHTML = '<span class="xx-role-empty">点击上方角色标签选中</span>';
    } else {
      sl.innerHTML = selectedRoles.map(tag =>
        `<span class="xx-role-item">${escapeHtml(tag)} <span class="xx-role-remove" data-tag="${tag}">✕</span></span>`
      ).join('');
    }

    // 渲染剧情标签
    const pt = document.getElementById('plotTags');
    pt.innerHTML = plotTagList.map((tag, i) => {
      const safeTag = escapeHtml(tag);
      return `<span class="xx-tag" data-type="plot" data-index="${i}" data-tag="${tag}">
        <span class="xx-tag-text">${safeTag}</span>
        <span class="xx-tag-edit" data-action="edit">✎</span>
        <span class="xx-tag-del" data-action="delete">×</span>
      </span>`;
    }).join('');

    // 恢复剧情草稿
    const pl = document.getElementById('pl');
    if (!pl.dataset.init) {
      const saved = localStorage.getItem('plotDraft');
      if (saved) pl.value = saved;
      pl.dataset.init = 1;
    }
    pl.oninput = function() {
      localStorage.setItem('plotDraft', this.value);
      if (!document.getElementById('pw').classList.contains('fullscreen')) autoResize(this);
    };
    autoResize(pl);
  }

  // -------- 事件委托（统一处理标签点击） --------
  document.getElementById('roleTags').addEventListener('click', function(e) {
    const target = e.target;
    const tagEl = target.closest('.xx-tag');
    if (!tagEl) return;
    const tag = tagEl.dataset.tag;
    const index = parseInt(tagEl.dataset.index);
    const type = tagEl.dataset.type;

    if (target.classList.contains('xx-tag-text')) {
      toggleRole(tag);
    } else if (target.classList.contains('xx-tag-edit')) {
      startEdit(type, index);
    } else if (target.classList.contains('xx-tag-del')) {
      if (type === 'role') deleteRoleTag(index);
      else deletePlotTag(index);
    }
  });

  document.getElementById('plotTags').addEventListener('click', function(e) {
    const target = e.target;
    const tagEl = target.closest('.xx-tag');
    if (!tagEl) return;
    const tag = tagEl.dataset.tag;
    const index = parseInt(tagEl.dataset.index);
    const type = tagEl.dataset.type;

    if (target.classList.contains('xx-tag-text')) {
      insertPlotTag(tag);
    } else if (target.classList.contains('xx-tag-edit')) {
      startEdit(type, index);
    } else if (target.classList.contains('xx-tag-del')) {
      if (type === 'role') deleteRoleTag(index);
      else deletePlotTag(index);
    }
  });

  // 已选角色列表的事件委托
  document.getElementById('selectedRoles').addEventListener('click', function(e) {
    const target = e.target;
    if (target.classList.contains('xx-role-remove')) {
      const tag = target.dataset.tag;
      removeSelected(tag);
    }
  });

  // -------- 内联编辑（自动聚焦 + 弹出键盘） --------
  function startEdit(type, index) {
    const list = type === 'role' ? roleTagList : plotTagList;
    const old = list[index];
    if (!old) return;

    const container = document.getElementById(type === 'role' ? 'roleTags' : 'plotTags');
    const tagEl = container.querySelector(`.xx-tag[data-type="${type}"][data-index="${index}"]`);
    if (!tagEl) return;

    const input = document.createElement('input');
    input.type = 'text';
    input.className = 'xx-tag-edit-input';
    input.value = old;
    tagEl.innerHTML = '';
    tagEl.appendChild(input);

    setTimeout(() => {
      input.focus();
      input.select();
    }, 50);

    input.addEventListener('keydown', function(e) {
      if (e.key === 'Enter') {
        e.preventDefault();
        finishEdit(type, index, input.value.trim());
      } else if (e.key === 'Escape') {
        e.preventDefault();
        cancelEdit(type, index);
      }
    });

    input.addEventListener('blur', function() {
      finishEdit(type, index, input.value.trim());
    }, { once: true });
  }

  function finishEdit(type, index, newVal) {
    const list = type === 'role' ? roleTagList : plotTagList;
    const old = list[index];
    if (newVal === '') {
      if (type === 'role') {
        deleteRoleTag(index);
      } else {
        deletePlotTag(index);
      }
      return;
    }
    if (newVal === old) {
      render();
      return;
    }
    list[index] = newVal;
    if (type === 'role') {
      const idx = selectedRoles.indexOf(old);
      if (idx > -1) selectedRoles[idx] = newVal;
      S('selectedRoles', selectedRoles);
      S('roleTags', roleTagList);
    } else {
      S('plotTags', plotTagList);
    }
    render();
  }

  function cancelEdit(type, index) {
    render();
  }

  // -------- 角色标签操作 --------
  window.toggleRole = function(tag) {
    const idx = selectedRoles.indexOf(tag);
    if (idx > -1) {
      selectedRoles.splice(idx, 1);
    } else {
      selectedRoles.push(tag);
    }
    S('selectedRoles', selectedRoles);
    render();
  };

  window.removeSelected = function(tag) {
    const idx = selectedRoles.indexOf(tag);
    if (idx > -1) selectedRoles.splice(idx, 1);
    S('selectedRoles', selectedRoles);
    render();
  };

  window.deleteRoleTag = function(i) {
    const tag = roleTagList[i];
    roleTagList.splice(i, 1);
    const idx = selectedRoles.indexOf(tag);
    if (idx > -1) selectedRoles.splice(idx, 1);
    S('roleTags', roleTagList);
    S('selectedRoles', selectedRoles);
    render();
  };

  // -------- 剧情标签操作 --------
  window.insertPlotTag = function(tag) {
    const pl = document.getElementById('pl');
    pl.value += tag + ' ';
    pl.dispatchEvent(new Event('input'));
    pl.focus();
  };

  window.deletePlotTag = function(i) {
    plotTagList.splice(i, 1);
    S('plotTags', plotTagList);
    render();
  };

  // -------- 添加标签（多行） --------
  document.getElementById('roleTagAdd').onclick = function() {
    const ta = document.getElementById('roleTagIn');
    const tags = parseTagsFromTextarea(ta);
    if (tags.length === 0) return alert('请输入至少一个标签');
    roleTagList.push(...tags);
    S('roleTags', roleTagList);
    render();
    ta.value = '';
    ta.focus();
  };

  document.getElementById('plotTagAdd').onclick = function() {
    const ta = document.getElementById('plotTagIn');
    const tags = parseTagsFromTextarea(ta);
    if (tags.length === 0) return alert('请输入至少一个标签');
    plotTagList.push(...tags);
    S('plotTags', plotTagList);
    render();
    ta.value = '';
    ta.focus();
  };

  // -------- 自动增高 --------
  function autoResize(el) {
    if (!el) return;
    el.style.height = 'auto';
    el.style.height = el.scrollHeight + 'px';
  }

  // -------- 全屏切换 --------
  const pw = document.getElementById('pw');
  const pt = document.getElementById('pt');
  let fs = false;
  pt.onclick = function(e) {
    e.stopPropagation();
    fs = !fs;
    pw.classList.toggle('fullscreen', fs);
    pt.classList.toggle('fullscreen-btn', fs);
    pt.textContent = fs ? '✕' : '⛶';
    if (fs) {
      const pl = document.getElementById('pl');
      setTimeout(() => { pl.focus(); pl.scrollTop = pl.scrollHeight; }, 200);
    } else {
      const pl = document.getElementById('pl');
      setTimeout(() => autoResize(pl), 50);
    }
  };
  document.addEventListener('keydown', e => { if (e.key === 'Escape' && fs) pt.click(); });
  document.getElementById('M').addEventListener('click', function(e) { if (e.target === this && fs) pt.click(); });
  document.querySelector('.xx-close').addEventListener('click', () => { if (fs) pt.click(); });

  // -------- 确定 --------
  window.confirmPlot = function() {
    const roles = selectedRoles.length ? selectedRoles.map((r, i) => `登场角色${i+1}：${r}`).join('\n') : '（无新登场角色）';
    const plot = document.getElementById('pl').value.trim();
    let out = '创作剧情\n' + roles + '\n\n本节剧情：' + plot;
    const ta = document.querySelector('.chat-input-scope textarea') || document.querySelector('.chatMsgTextarea textarea') || document.querySelector('textarea');
    if (ta) {
      ta.value = out;
      ta.dispatchEvent(new Event('input', { bubbles: true }));
      document.getElementById('M').classList.remove('show');
    }
  };

  // -------- 启动 --------
  render();
</script>
