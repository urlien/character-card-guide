<script>
    // ===== 全局变量声明 =====
    var currentRevPage = 1;
    var currentStage = 0;          // 当前显示的阶段索引 (0~6)
    var isMenuOpen = false;
    var isDragging = false;
    var startX, startY, originLeft, originTop;

    // ===== 宣传册翻页变量 =====
    var brochurePage = 1;

    // ===== 修复后的拖动逻辑 =====
    function dragStart(e) {
        if (e.target && e.target.classList && e.target.classList.contains('sub-btn')) {
            return;
        }
        e.preventDefault();
        var touch = e.touches ? e.touches[0] : e;
        startX = touch.clientX;
        startY = touch.clientY;

        var menu = document.getElementById('SideMenu');
        var rect = menu.getBoundingClientRect();
        originLeft = rect.left;
        originTop = rect.top;

        isDragging = false;
        document.addEventListener('mousemove', dragMove);
        document.addEventListener('touchmove', dragMove, {passive: false});
        document.addEventListener('mouseup', dragEnd);
        document.addEventListener('touchend', dragEnd);
    }

    function dragMove(e) {
        e.preventDefault();
        var touch = e.touches ? e.touches[0] : e;
        var dx = touch.clientX - startX;
        var dy = touch.clientY - startY;

        if (Math.abs(dx) > 5 || Math.abs(dy) > 5) {
            isDragging = true;
        }

        if (isDragging) {
            var menu = document.getElementById('SideMenu');
            var newLeft = Math.max(0, Math.min(window.innerWidth - 50, originLeft + dx));
            var newTop = Math.max(0, Math.min(window.innerHeight - 50, originTop + dy));
            menu.style.left = newLeft + 'px';
            menu.style.top = newTop + 'px';
        }
    }

    function dragEnd(e) {
        document.removeEventListener('mousemove', dragMove);
        document.removeEventListener('touchmove', dragMove);
        document.removeEventListener('mouseup', dragEnd);
        document.removeEventListener('touchend', dragEnd);

        if (!isDragging) {
            toggleMenu();
        }
    }


    // ===== 基础逻辑控制 =====
    function checkSeen() {
        try {
            var seen = localStorage.getItem('pasture_seen');
            if (!seen) {
                startIntro();
                localStorage.setItem('pasture_seen', 'true');
            }
        } catch(e) { 
            startIntro(); 
        }
    }

    function toggleMenu() {
        isMenuOpen = !isMenuOpen;
        var menu = document.getElementById('SideMenu');
        var btn = document.getElementById('MainBtn');
        if (isMenuOpen) {
            menu.classList.add('open');
            btn.classList.add('open');
        } else {
            menu.classList.remove('open');
            btn.classList.remove('open');
        }
    }

    function replayAnimation() {
        toggleMenu(); 
        startIntro();
    }

    // ===== 隐藏侧边栏（仅临时隐藏，刷新后恢复显示） =====
    function hideSideMenu() {
        // 如果菜单展开则先关闭
        if (isMenuOpen) {
            toggleMenu();
        }
        // 直接隐藏整个 SideMenu（不存 localStorage，刷新即恢复）
        document.getElementById('SideMenu').style.display = 'none';
    }

    // ===== 开场动画控制（点击切换，共7阶段） =====
    function startIntro() {
        var overlay = document.getElementById('Charm_Promo');
        overlay.style.display = 'flex';
        currentStage = 0;
        showStage(0);
    }

    function showStage(index) {
        // 隐藏所有阶段（1~7）
        for (var i = 1; i <= 7; i++) {
            document.getElementById('stage' + i).style.display = 'none';
        }
        // 显示当前阶段
        var stageId = 'stage' + (index + 1);
        document.getElementById(stageId).style.display = 'flex';

        // 更新底部提示文字（最后一页为阶段7，索引6）
        var hint = document.getElementById('clickHint');
        if (index === 6) {
            hint.innerHTML = '✦ 点击进入魅魔牧场 ✦';
        } else {
            hint.innerHTML = '✦ 点击任意处继续 ✦';
        }
    }

    function nextIntroStage() {
        // 如果当前是最后一页（索引6），则关闭开场
        if (currentStage === 6) {
            document.getElementById('Charm_Promo').style.display = 'none';
            return;
        }
        // 否则切换到下一页
        currentStage++;
        showStage(currentStage);
    }

    // ===== 宣传册翻页逻辑 =====
    function changeBrochurePage(delta) {
        var newPage = brochurePage + delta;
        if (newPage < 1 || newPage > 3) return;
        brochurePage = newPage;
        // 隐藏所有页面
        for (var i = 1; i <= 3; i++) {
            document.getElementById('brochurePage' + i).style.display = 'none';
        }
        document.getElementById('brochurePage' + brochurePage).style.display = 'block';
        document.getElementById('brochurePageIndicator').innerText = brochurePage + ' / 3';
    }

    // ===== 回顾/说明书 =====
    function openReview() {
        toggleMenu();
        document.getElementById('Charm_Review').style.display = 'flex';
        currentRevPage = 1;
        updateReview();
    }

    function closeReview() {
        document.getElementById('Charm_Review').style.display = 'none';
    }

    function updateReview() {
        for (var i = 1; i <= 4; i++) document.getElementById('revStep'+i).style.display = 'none';
        document.getElementById('revStep'+currentRevPage).style.display = 'block';
        document.getElementById('revProg').innerText = currentRevPage + ' / 4';
    }

    function revNext() {
        if (currentRevPage < 4) {
            currentRevPage++;
            updateReview();
        } else {
            closeReview();
        }
    }

    function revPrev() {
        if (currentRevPage > 1) {
            currentRevPage--;
            updateReview();
        }
    }

    // ===== 直接初始化 =====
    (function initButton() {
        // 每次刷新都正常显示（不检查隐藏状态）
        if (!document.getElementById('SideMenu').style.left) {
            document.getElementById('SideMenu').style.left = '85%';
            document.getElementById('SideMenu').style.top = '80%';
        }
        checkSeen();
    })();
</script>
