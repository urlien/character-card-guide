function getRandomQuote(){
    var idx = Math.floor(Math.random() * window.quotes.length);
    return window.quotes[idx];
}

function getEventQuote(event){
    if(event && event.q) return event.q;
    return getRandomQuote();
}

function updateQuote(text){
    var bubble = document.getElementById('event-quote');
    if(bubble){
        bubble.style.opacity = '0';
        setTimeout(function(){
            bubble.innerText = text;
            bubble.style.opacity = '1';
        }, 100);
    }
}

function refreshRandomQuote(){
    updateQuote(getRandomQuote());
}

function updateAvatar(){
    var img = document.getElementById('event-avatar-img');
    var placeholder = document.querySelector('.event-avatar-placeholder');
    if(img){
        img.src = window.avatarUrl + '?t=' + Date.now();
        img.style.display = 'block';
        if(placeholder) placeholder.style.display = 'none';
    }
}

function updateDisplay(){
    var nameEl = document.getElementById('event-name');
    var descEl = document.getElementById('event-desc');
    if(!nameEl||!descEl) return;
    if(window.currentEvent){
        nameEl.innerText = window.currentEvent.n;
        descEl.innerText = window.currentEvent.d;
        updateQuote(getEventQuote(window.currentEvent));
    } else {
        nameEl.innerText = '未抽取';
        descEl.innerText = '';
        updateQuote('✨ 点击「随机抽取」或点击头像，命运将为你揭示下一个事件 ✨');
    }
}

function drawEvent(){
    var type = window.currentType;
    var events = window.eventData[type];
    if(!events||events.length===0) return;
    var idx = Math.floor(Math.random() * events.length);
    window.currentEvent = events[idx];
    window.currentIndex = idx;
    updateDisplay();
    var avatar = document.getElementById('event-avatar');
    if(avatar){
        avatar.style.transform = 'scale(0.95)';
        setTimeout(function(){if(avatar)avatar.style.transform = '';},200);
    }
}

function nextEvent(){
    var type = window.currentType;
    var events = window.eventData[type];
    if(!events||events.length===0) return;
    if(!window.currentEvent){
        window.currentIndex = 0;
        window.currentEvent = events[0];
    } else {
        window.currentIndex = (window.currentIndex + 1) % events.length;
        window.currentEvent = events[window.currentIndex];
    }
    updateDisplay();
}

function prevEvent(){
    var type = window.currentType;
    var events = window.eventData[type];
    if(!events||events.length===0) return;
    if(!window.currentEvent){
        window.currentIndex = events.length - 1;
        window.currentEvent = events[window.currentIndex];
    } else {
        window.currentIndex = (window.currentIndex - 1 + events.length) % events.length;
        window.currentEvent = events[window.currentIndex];
    }
    updateDisplay();
}

function switchType(type){
    window.currentType = type;
    window.currentEvent = null;
    var btns = document.querySelectorAll('.event-type-btn');
    for(var i=0;i<btns.length;i++){
        if(btns[i].getAttribute('data-type')===type){
            btns[i].classList.add('active');
        } else {
            btns[i].classList.remove('active');
        }
    }
    updateDisplay();
    refreshRandomQuote();
}

function injectEvent(){
    if(!window.currentEvent){
        updateQuote('⚠️ 请先随机抽取一个事件！');
        setTimeout(function(){if(window.currentEvent)updateQuote(getEventQuote(window.currentEvent));else updateQuote(getRandomQuote());},1500);
        return;
    }
    var ta = document.querySelector('.chat .chat-bottom .uni-textarea .chat-input-scope textarea') || document.querySelector('#send_textarea') || document.querySelector('textarea[placeholder*="输入"]') || document.querySelector('.chatMsgTextarea textarea');
    if(ta){
        var injectText = '生成事件：【' + window.currentEvent.n + '】\n' + window.currentEvent.d;
        var oldVal = ta.value;
        ta.value = oldVal + (oldVal ? '\n\n' : '') + injectText;
        ta.dispatchEvent(new Event('input',{bubbles:true}));
        ta.dispatchEvent(new Event('change',{bubbles:true}));
        if(ta.focus) ta.focus();
        var useBtn = document.getElementById('event-use');
        if(useBtn){
            var originalText = useBtn.innerText;
            useBtn.innerText = '✓ 已启用';
            setTimeout(function(){
                if(useBtn) useBtn.innerText = originalText;
            },1200);
        }
        updateQuote('✨ 事件已注入！期待接下来的发展吧 ✨');
        setTimeout(function(){
            var modal = document.getElementById('event-modal');
            if(modal) modal.classList.remove('active');
        },800);
    } else {
        updateQuote('⚠️ 未找到输入框，请手动复制事件内容。');
        setTimeout(function(){
            if(window.currentEvent) updateQuote(getEventQuote(window.currentEvent));
            else updateQuote(getRandomQuote());
        },2000);
    }
}
