  <!-- 修复版JS：强初始化+防重复+ID绑定 -->
  <script>
    (function() {
      // 全局状态锁，防止重复初始化
      if (window.statusBarInitialized) return;
      window.statusBarInitialized = true;

      // ====================== 配置 ======================
      const CONFIG = {
        regex: /[#＃]状态栏开始[#＃]\s*(?:.*?\n)?(?:▣\s*剧情节点\/日期\/时段[ :：\s]*(.*?)\s*◆\s*位置[ :：\s]*(.*?)\s*■\s*体力[ :：\s]*(.*?)\s*✦\s*心绪[ :：\s]*(.*?)\s*➤\s*可前往(?:\/当前已解锁地点)?[ :：\s]*(.*?)\s*▶\s*当前动作[ :：\s]*(.*?)\s*▧\s*着装[ :：\s]*(.*?)\s*✧\s*变装[ :：\s]*(.*?)\s*⚔\s*武器[ :：\s]*(.*?)\s*▣\s*物品[ :：\s]*(.*?)\s*✦\s*特长[ :：\s]*(.*?)\s*➜\s*(?:未来)?规划[ :：\s]*(.*?)\s*💰\s*货币[ :：\s]*(.*?)\s*💗\s*好感度[ :：\s]*(.*?)\s*💭\s*内心独白[ :：\s]*(.*?)\s*📰\s*(?:城市)?快报1[ :：\s]*(.*?)\s*📰\s*(?:城市)?快报2[ :：\s]*(.*?)\s*📰\s*(?:城市)?快报3[ :：\s]*(.*?)\s*▣\s*当前轮次[ :：\s]*(.*?)\s*|(.*?)\s*✦\s*外貌[ :：\s]*(.*?)\s*▧\s*衣装[ :：\s]*(.*?)\s*▶\s*姿势[ :：\s]*(.*?)\s*✦\s*(.*?)\s*■\s*(.*?)\s*♡\s*(.*?)\s*▣\s*(.*?)\s*💗\s*好感度[ :：\s]*(.*?)\s*✧\s*心绪[ :：\s]*(.*?)\s*)\s*[#＃]状态栏结束[#＃]/s,
        totalPages: 3
      };

      // ====================== DOM元素（ID绑定，绝对可靠） ======================
      const elements = {
        pages: [
          document.getElementById('page-1'),
          document.getElementById('page-2'),
          document.getElementById('page-3')
        ],
        prevBtn: document.getElementById('prev-btn'),
        nextBtn: document.getElementById('next-btn'),
        pageIndicator: document.getElementById('page-indicator')
      };

      // ====================== 翻页逻辑 ======================
      let currentPage = 1;

      function updatePage() {
        // 隐藏所有页面
        elements.pages.forEach(page => {
          if (page) page.classList.remove('active');
        });

        // 显示当前页面
        const activePage = elements.pages[currentPage - 1];
        if (activePage) activePage.classList.add('active');

        // 更新指示器和按钮状态
        elements.pageIndicator.textContent = `${currentPage}/${CONFIG.totalPages}`;
        elements.prevBtn.disabled = currentPage === 1;
        elements.nextBtn.disabled = currentPage === CONFIG.totalPages;
      }

      // 绑定翻页事件
      elements.prevBtn.addEventListener('click', () => {
        if (currentPage > 1) {
          currentPage--;
          updatePage();
        }
      });

      elements.nextBtn.addEventListener('click', () => {
        if (currentPage < CONFIG.totalPages) {
          currentPage++;
          updatePage();
        }
      });

      // ====================== 解析填充逻辑 ======================
      function parseStatus(text) {
        if (!text || typeof text !== 'string') return [];
        
        const results = [];
        let match;
        const regex = new RegExp(CONFIG.regex.source, 'gs');
        
        while ((match = regex.exec(text)) !== null) {
          if (match[1]) {
            results.push({
              type: 'player',
              data: match.slice(1, 20).map(v => v?.trim() || '')
            });
          } else if (match[20]) {
            results.push({
              type: 'npc',
              data: match.slice(20, 30).map(v => v?.trim() || '')
            });
          }
        }
        
        return results;
      }

      function fillStatus(statusList) {
        if (!Array.isArray(statusList) || statusList.length === 0) return false;

        // 清空所有值，防止残留
        for (let i = 1; i <= 31; i++) {
          const el = document.getElementById(`val-${i}`);
          if (el) el.textContent = '';
        }

        // 按顺序填充
        let lastType = 'player';
        statusList.forEach(status => {
          if (status.type === 'player') {
            // 主角：填充1-19
            for (let i = 0; i < 19; i++) {
              const el = document.getElementById(`val-${i + 1}`);
              if (el) el.textContent = status.data[i] || '';
            }
            lastType = 'player';
          } else if (status.type === 'npc') {
            // NPC：填充20-29
            for (let i = 0; i < 10; i++) {
              const el = document.getElementById(`val-${20 + i}`);
              if (el) el.textContent = status.data[i] || '';
            }
            lastType = 'npc';
          }
        });

        // 自动跳转到最后一个状态栏的页面
        currentPage = lastType === 'player' ? 1 : 2;
        updatePage();

        return true;
      }

      // ====================== 自动解析当前页面的状态栏 ======================
      function autoParseCurrentPage() {
        // 获取当前状态栏所在的整个消息内容
        const container = document.querySelector('.teapot-status-container');
        if (!container) return;

        // 查找父消息元素
        let messageEl = container.closest('.message, .chat-message, .ai-message');
        if (!messageEl) return;

        // 解析并填充
        const statusList = parseStatus(messageEl.textContent);
        if (statusList.length > 0) {
          fillStatus(statusList);
          console.log(`自动解析成功：${statusList.length} 个状态栏`);
        }
      }

      // ====================== 初始化 ======================
      // 确保DOM完全加载后执行
      if (document.readyState === 'loading') {
        document.addEventListener('DOMContentLoaded', () => {
          updatePage();
          setTimeout(autoParseCurrentPage, 100); // 延迟100ms确保渲染完成
        });
      } else {
        updatePage();
        setTimeout(autoParseCurrentPage, 100);
      }

      // 暴露全局方法
      window.statusParser = {
        parse: parseStatus,
        fill: fillStatus,
        goToPage: (page) => {
          if (page >= 1 && page <= CONFIG.totalPages) {
            currentPage = page;
            updatePage();
          }
        }
      };
    })();
  </script>
</div>
