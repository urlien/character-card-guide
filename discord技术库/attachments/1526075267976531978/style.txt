  const multiKeys = ['event', 'trait', 'mod'];
  document.querySelectorAll('.option-grid').forEach(grid => {
    grid.addEventListener('click', function(e) {
      const d = e.target.closest('.d');
      if (!d) return;
      const key = grid.id.replace('grid-', '');
      const val = d.dataset.t;
      if (multiKeys.includes(key)) {
        d.classList.toggle('selected');
        brainleaf.toggleMulti(key, val);
      } else {
        grid.querySelectorAll('.d').forEach(el => el.classList.remove('selected'));
        d.classList.add('selected');
        brainleaf.set(key, val);
      }
    });
  });
})();
</script>
